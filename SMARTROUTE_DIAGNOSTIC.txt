  18: def haversine_distance_km(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
  19:     """Calculates great-circle distance in kilometers between two points."""
  20:     r = 6371.0  # Earth radius in km
  21:     phi1 = math.radians(lat1)
  22:     phi2 = math.radians(lat2)
  23:     delta_phi = math.radians(lat2 - lat1)
  24:     delta_lambda = math.radians(lon2 - lon1)
  25: 
  26:     a = math.sin(delta_phi / 2.0) ** 2 + math.cos(phi1) * math.cos(phi2) * math.sin(delta_lambda / 2.0) ** 2
  27:     c = 2.0 * math.atan2(math.sqrt(a), math.sqrt(1.0 - a))
  28:     return r * c

  31: def generate_curved_geometry(
  32:     start: Tuple[float, float],
  33:     end: Tuple[float, float],
  34:     deviation_factor: float = 0.0,
  35:     steps: int = 40
  36: ) -> List[List[float]]:
  37:     """
  38:     Generates a realistic multi-point highway geometry connecting start and end
  39:     with controlled lateral curve deviation to simulate alternative road corridors.
  40:     Returns list of [latitude, longitude] points for route-map clients.
  41:     """
  42:     lat1, lon1 = start
  43:     lat2, lon2 = end
  44:     coords = []
  45: 
  46:     # Perpendicular vector for lateral highway arc
  47:     dlat = lat2 - lat1
  48:     dlon = lon2 - lon1
  49:     perp_lat = -dlon * deviation_factor
  50:     perp_lon = dlat * deviation_factor
  51: 
  52:     for i in range(steps + 1):
  53:         t = i / float(steps)
  54:         # Base linear interpolation
  55:         base_lat = lat1 + t * dlat
  56:         base_lon = lon1 + t * dlon
  57:         # Parabolic lateral offset
  58:         curve = math.sin(t * math.pi)
  59:         # Micro-variation to look like real roads rather than rigid math arcs
  60:         micro_lat = math.sin(t * 8.0 * math.pi) * 0.015 * abs(dlat + 0.1)
  61:         micro_lon = math.cos(t * 8.0 * math.pi) * 0.015 * abs(dlon + 0.1)
  62: 
  63:         pt_lat = round(base_lat + (perp_lat * curve) + micro_lat, 5)
  64:         pt_lon = round(base_lon + (perp_lon * curve) + micro_lon, 5)
  65:         coords.append([pt_lat, pt_lon])
  66: 
  67:     return coords
